   138→function kelasStatusRingkas(s: string): { bg: string; text: string; label: string } | null {
   139→  if (s === "Hadir")        return { bg: WARNA_STATUS.Hadir.bg,        text: WARNA_STATUS.Hadir.text,        label: WARNA_STATUS.Hadir.label };
   140→  if (s === "Lewat")        return { bg: WARNA_STATUS.Lewat.bg,        text: WARNA_STATUS.Lewat.text,        label: WARNA_STATUS.Lewat.label };
   141→  if (s === "TidakHadir")   return { bg: WARNA_STATUS.TidakHadir.bg,   text: WARNA_STATUS.TidakHadir.text,   label: WARNA_STATUS.TidakHadir.label };
   142→  if (s === "Cuti")         return { bg: WARNA_STATUS.Cuti.bg,         text: WARNA_STATUS.Cuti.text,         label: WARNA_STATUS.Cuti.label };
   143→  if (s === "TugasRasmi")   return { bg: WARNA_STATUS.TugasRasmi.bg,   text: WARNA_STATUS.TugasRasmi.text,   label: WARNA_STATUS.TugasRasmi.label };
   144→  if (s === "Cuti Umum" || s === "CutiUmum") return { bg: WARNA_STATUS.CutiUmum.bg, text: WARNA_STATUS.CutiUmum.text, label: WARNA_STATUS.CutiUmum.label };
   145→  return null;
   146→}

   394→function TabHarian(props: {
   395→  tarikh: string;
   396→  setTarikh: (v: string) => void;
   397→  jabatanId: string;
   398→  senJabatan: JabatanRingkas[];
   399→  setJabatanId: (v: string) => void;
   400→  kategori: string;
   401→  setKategori: (v: string) => void;
   402→  setKategoriIds: Set<string> | null;
   403→}) {
   404→  const { tarikh, setTarikh, jabatanId, senJabatan, setJabatanId, kategori, setKategori, setKategoriIds } = props;
   405→  const [data, setData] = useState<ResponsHarian | null>(null);
   406→  const [memuat, setMemuat] = useState(true);
   407→
   408→  const muat = useCallback(async () => {
   409→    if (!tarikh) return;
   410→    setMemuat(true);
   411→    try {
   412→      const qs = new URLSearchParams({ tarikh });
   413→      if (jabatanId !== "__SEMUA__") qs.set("jabatanId", jabatanId);
   414→      const r = await apiFetch<ResponsHarian>(`/api/laporan/harian?${qs.toString()}`);
   415→      setData(r);
   416→    } catch (err) {
   417→      toast.error(err instanceof Error ? err.message : "Gagal memuatkan laporan harian.");
   418→      setData(null);
   419→    } finally {
   420→      setMemuat(false);
   421→    }
   422→  }, [tarikh, jabatanId]);
   423→
   424→  useEffect(() => { muat(); }, [muat]);
   425→
   426→  const barisTapis = useMemo(() => {
   427→    if (!data) return [];
   428→    if (!setKategoriIds) return data.senarai;
   429→    return data.senarai.filter((b) => setKategoriIds.has(b.staf.id));
   430→  }, [data, setKategoriIds]);
   431→
   432→  function eksport() {
   433→    if (!data) return;
   434→    const lajur = ["NoIdStaf", "Nama", "Jawatan", "Jabatan", "MasaMasuk", "MasaKeluar", "JamBekerja", "Status"];
   435→    const baris = barisTapis.map((b) => ({
   436→      NoIdStaf: b.staf.noIdStaf,
   437→      Nama: b.staf.nama,
   438→      Jawatan: b.staf.jawatan,
   439→      Jabatan: b.staf.jabatan,
   440→      MasaMasuk: formatMasa(b.kehadiran?.masaMasuk),
   441→      MasaKeluar: formatMasa(b.kehadiran?.masaKeluar),
   442→      JamBekerja: b.kehadiran?.jamBekerja != null ? b.kehadiran.jamBekerja.toFixed(2) : "",
   443→      Status: b.statusRingkas,
   444→    }));
   445→    eksportCSV(`laporan-harian-${tarikh}.csv`, lajur, baris);
   446→  }
   447→
   448→  const tarikhDate = tarikh ? new Date(tarikh + "T00:00:00") : null;
   449→  const namaHari = tarikhDate
   450→    ? ["Ahad", "Isnin", "Selasa", "Rabu", "Khamis", "Jumaat", "Sabtu"][tarikhDate.getDay()]
   451→    : "";
   452→
   453→  return (
   454→    <div className="space-y-4">
   455→      <div className="skeuo-card p-4">
   456→        <BarisPenapis>
   457→          <div>
   458→            <Label className="text-xs font-semibold text-stone-600">Tarikh</Label>
   459→            <Input type="date" value={tarikh} onChange={(e) => setTarikh(e.target.value)} className="skeuo-input mt-1" />
   460→          </div>
   461→          <PilihJabatan senarai={senJabatan} value={jabatanId} onChange={setJabatanId} />
   462→          <PilihKategori value={kategori} onChange={setKategori} />
   463→          <div className="flex items-end gap-2">
   464→            <Button onClick={eksport} disabled={!data || barisTapis.length === 0} className="skeuo-btn-secondary h-9 gap-1.5">
   465→              <Download size={15} /> Eksport CSV
   466→            </Button>
   467→          </div>
   468→        </BarisPenapis>
   469→      </div>
   470→
   471→      <div className="skeuo-card p-4">
   472→        <TajukSekolah tetapan={data?.tetapan} sub={`Laporan Harian Kehadiran — ${namaHari}, ${formatTarikhBM(tarikh)}`} />
   473→        <div className="flex items-center justify-between mb-3 print:hidden">
   474→          <h3 className="font-bold text-stone-800 text-sm flex items-center gap-1.5">
   475→            <CalendarDays size={15} className="text-stone-500" />
   476→            {namaHari}, {formatTarikhBM(tarikh)}
   477→          </h3>
   478→          <div className="flex items-center gap-2">
   479→            {data?.isCutiUmum && (
   480→              <span className="skeuo-pill bg-slate-200 text-slate-700">Cuti Umum: {data.cutiUmum}</span>
   481→            )}
   482→            {data?.isHujungMinggu && (
   483→              <span className="skeuo-pill bg-slate-200 text-slate-700">Hujung Minggu</span>
   484→            )}

  1022→function TabCuti(props: {
  1023→  tahun: number;
  1024→  setTahun: (v: number) => void;
  1025→  jabatanId: string;
  1026→  senJabatan: JabatanRingkas[];
  1027→  setJabatanId: (v: string) => void;
  1028→  kategori: string;
  1029→  setKategori: (v: string) => void;
  1030→  setKategoriIds: Set<string> | null;
  1031→}) {
  1032→  const { tahun, setTahun, jabatanId, senJabatan, setJabatanId, kategori, setKategori, setKategoriIds } = props;
  1033→  const [data, setData] = useState<ResponsCuti | null>(null);
  1034→  const [memuat, setMemuat] = useState(true);
  1035→
  1036→  const muat = useCallback(async () => {
  1037→    setMemuat(true);
  1038→    try {
  1039→      const qs = new URLSearchParams({ tahun: String(tahun) });
  1040→      if (jabatanId !== "__SEMUA__") qs.set("jabatanId", jabatanId);
  1041→      const r = await apiFetch<ResponsCuti>(`/api/laporan/cuti?${qs.toString()}`);
  1042→      setData(r);
  1043→    } catch (err) {
  1044→      toast.error(err instanceof Error ? err.message : "Gagal memuatkan laporan cuti.");
  1045→      setData(null);
  1046→    } finally {
  1047→      setMemuat(false);
  1048→    }
  1049→  }, [tahun, jabatanId]);
  1050→
  1051→  useEffect(() => { muat(); }, [muat]);
  1052→
  1053→  // Senarai jenis cuti unik
  1054→  const senJenis = useMemo(() => {
  1055→    if (!data) return [] as { id: string; nama: string }[];
  1056→    const m = new Map<string, string>();
  1057→    for (const b of data.senarai) for (const bc of b.bakiCuti) {
  1058→      if (!m.has(bc.jenisCutiId)) m.set(bc.jenisCutiId, namaJenisCuti(bc.jenisCuti));
  1059→    }
  1060→    return Array.from(m.entries()).map(([id, nama]) => ({ id, nama }));
  1061→  }, [data]);
  1062→
  1063→  const barisTapis = useMemo(() => {
  1064→    if (!data) return [];
  1065→    if (!setKategoriIds) return data.senarai;
  1066→    return data.senarai.filter((b) => setKategoriIds.has(b.stafId));
  1067→  }, [data, setKategoriIds]);
  1068→
  1069→  function bakiStaf(baris: BarisCuti, jenisCutiId: string): { layak: number; digunakan: number; baki: number } {
  1070→    const entri = baris.bakiCuti.find((b) => b.jenisCutiId === jenisCutiId);
  1071→    if (!entri) return { layak: 0, digunakan: 0, baki: 0 };
  1072→    return {
  1073→      layak: entri.layak,
  1074→      digunakan: entri.digunakan,
  1075→      baki: entri.baki != null ? entri.baki : entri.layak - entri.digunakan,
  1076→    };
  1077→  }
  1078→
  1079→  function eksport() {
  1080→    if (!data) return;
  1081→    const lajur = ["Nama", "Jabatan", ...senJenis.map((j) => j.nama + " (digunakan)"), ...senJenis.map((j) => j.nama + " (baki)")];
  1082→    const baris = barisTapis.map((b) => {
  1083→      const row: Record<string, unknown> = { Nama: b.nama, Jabatan: b.jabatan };
  1084→      for (const j of senJenis) {
  1085→        const v = bakiStaf(b, j.id);
  1086→        row[`${j.nama} (digunakan)`] = v.digunakan;
  1087→      }
  1088→      for (const j of senJenis) {
  1089→        const v = bakiStaf(b, j.id);
  1090→        row[`${j.nama} (baki)`] = v.baki;
  1091→      }
  1092→      return row;
  1093→    });
  1094→    eksportCSV(`laporan-cuti-${tahun}.csv`, lajur, baris);
  1095→  }
  1096→
  1097→  const senTahun = useMemo(() => {
  1098→    const y = new Date().getFullYear();
  1099→    return [y - 1, y, y + 1];
  1100→  }, []);
  1101→
  1102→  return (
  1103→    <div className="space-y-4">
  1104→      <div className="skeuo-card p-4">
  1105→        <BarisPenapis>
  1106→          <div>
  1107→            <Label className="text-xs font-semibold text-stone-600">Tahun</Label>
  1108→            <Select value={String(tahun)} onValueChange={(v) => setTahun(Number(v))}>
  1109→              <SelectTrigger className="skeuo-input mt-1"><SelectValue /></SelectTrigger>
  1110→              <SelectContent>
  1111→                {senTahun.map((t) => (<SelectItem key={t} value={String(t)}>{t}</SelectItem>))}
  1112→              </SelectContent>
  1113→            </Select>
  1114→          </div>
  1115→          <PilihJabatan senarai={senJabatan} value={jabatanId} onChange={setJabatanId} />
  1116→          <PilihKategori value={kategori} onChange={setKategori} />
  1117→          <div className="flex items-end gap-2">
  1118→            <Button onClick={eksport} disabled={!data || barisTapis.length === 0} className="skeuo-btn-secondary h-9 gap-1.5">
  1119→              <Download size={15} /> CSV
  1120→            </Button>
  1121→          </div>
  1122→        </BarisPenapis>
  1123→      </div>
  1124→
  1125→      <div className="skeuo-card p-4">
  1126→        <h3 className="font-bold text-stone-800 text-sm flex items-center gap-1.5 mb-3 print:hidden">
  1127→          <FileText size={15} className="text-stone-500" /> Laporan Baki Cuti — Tahun {tahun}
  1128→          <span className="skeuo-pill bg-stone-200 text-stone-700 ml-2">{barisTapis.length} staf</span>
  1129→        </h3>
  1130→        <div className="text-center hidden print:block mb-3">
  1131→          <div className="text-base font-bold text-stone-800">SRI Al-Amin Gombak</div>
  1132→          <div className="text-sm font-semibold text-stone-700 mt-1">Laporan Baki Cuti — Tahun {tahun}</div>
  1133→        </div>
  1134→
  1135→        {memuat ? (
  1136→          <SkeletonJadual />
  1137→        ) : barisTapis.length === 0 ? (
  1138→          <KeadaanKosong mesej="Tiada data cuti dijumpai untuk penapis ini." />
  1139→        ) : (
  1140→          <div className="max-h-[34rem] overflow-auto custom-scroll">
  1141→            <Table>
  1142→              <TableHeader>
  1143→                <TableRow>
  1144→                  <TableHead className="text-xs sticky left-0 bg-stone-50 min-w-[160px]">Staf</TableHead>
  1145→                  <TableHead className="text-xs min-w-[120px]">Jawatan</TableHead>
